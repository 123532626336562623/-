      function showToast(message, emoji) {

        toastEl.classList.add('show');
        if (toastTimer) clearTimeout(toastTimer);
        toastTimer = setTimeout(function() {
          toastEl.classList.remove('show');
        }, 2200);
      }

      function triggerGreet(e) {
        var cx = (e.clientX || e.touches && e.touches[0].clientX) || window.innerWidth / 2;
        var cy = (e.clientY || e.touches && e.touches[0].clientY) || window.innerHeight / 2;
        fireConfetti(cx, cy);
        showToast('你好呀！很高兴见到你 👋', '🌟');
      }

      function triggerCopyEmail(e) {
        var email = '2175679295@qq.com';
        var doCopy = function() {
          var cx = (e.clientX || e.touches && e.touches[0].clientX) || window.innerWidth / 2;
          var cy = (e.clientY || e.touches && e.touches[0].clientY) || window.innerHeight / 2;
          fireConfetti(cx, cy);
          showToast('邮箱已复制到剪贴板 📋', '📧');
        };
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(email).then(doCopy).catch(function() {
            fallbackCopy(email, e);
          });
        } else {
          fallbackCopy(email, e);
        }
      }

      function fallbackCopy(text, e) {
        var textarea = document.createElement('textarea');
        textarea.value = text;
        textarea.style.position = 'fixed';
        textarea.style.opacity = '0';
        textarea.style.left = '-9999px';
        document.body.appendChild(textarea);
        textarea.select();
        try {
          document.execCommand('copy');
          var cx = e.clientX || window.innerWidth / 2;
          var cy = e.clientY || window.innerHeight / 2;
          fireConfetti(cx, cy);
          showToast('邮箱已复制到剪贴板 📋', '📧');
        } catch (err) {
          showToast('复制失败，请手动复制 😅', '⚠️');
        }
        document.body.removeChild(textarea);
      }

      var greetBtns = document.querySelectorAll('#btn-greet, #btn-greet-2');
      var copyBtns = document.querySelectorAll('#btn-copy-email, #btn-copy-email-2');
      greetBtns.forEach(function(btn) { btn.addEventListener('click', triggerGreet); });
      copyBtns.forEach(function(btn) { btn.addEventListener('click', triggerCopyEmail); });

        cards.forEach(function(card) {
          setTimeout(function() {
            if (!card.classList.contains('is-visible')) {
              card.classList.add('is-visible');
            }
          }, 1500);
        });
        progressBar.style.width = '0%';
      });

      window.addEventListener('beforeunload', function() {
        if (animId) cancelAnimationFrame(animId);
        if (confettiAnimId) cancelAnimationFrame(confettiAnimId);
        if (timer) clearTimeout(timer);
      });

    })();
